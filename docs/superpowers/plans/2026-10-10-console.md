# fdc console Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A live terminal screen, `fdc console`, that shows what the collector runs, where its data comes from and when each run happened, and can start the sync or one step from a key press.

**Architecture:** A read-only model module assembles an `Overview` dataclass from `Store` methods, a heartbeat file the running sync keeps current, and (on Windows) the Task Scheduler's registration of the sync task. A Textual app renders that model on a timer and starts `fdc sync` as a subprocess; a rich report renders the same model once for `--once` and non-terminal output. The sync gains an `origin` per run and waits its turn when another sync is alive.

**Tech Stack:** Python 3.11, SQLite (WAL), rich 15, Textual 8 (new dependency), pytest. No pytest plugins: Textual tests run under `asyncio.run`.

**Spec:** `docs/superpowers/specs/2026-10-10-console-design.md`

## Global Constraints

- Only `store.py` (and `migrate.py`, `readonly.py`) execute SQL. The model calls `Store` methods; the app and the report hold no SQL.
- Schema changes are migrations: `migrations/0008_run_origin.sql`; never edit an applied file.
- No network, no real scheduler, no real key store and no sleeping in tests. Collaborators (`run`, `popen`, `sleep`, `now`, `alive`, `store_open`) are parameters.
- Privacy: fixtures are synthetic; no real label, balance, symbol list or key in code, tests, docs or commit messages. `.json` files may only live under `tests/fixtures/`. `scripts/*.ps1` stay pure ASCII without a BOM (`tests/test_privacy.py`).
- `data/` stays gitignored; the heartbeat (`data/sync.heartbeat.json`) and the console's run log (`data/console-last-run.log`) live there.
- The console never writes to the database. `fdc console` opens the store once at start to apply pending migrations, as every `fdc` command does; every refresh afterwards opens with `migrate=False` and only reads.
- Never `os.kill(pid, 0)` on Windows (it terminates the process). Liveness goes through `heartbeat.pid_alive`.
- Dependency floor: `textual>=8`. Version in `pyproject.toml` stays `0.3.0`; notes go under `## Unreleased` in `CHANGELOG.md`.
- Display strings stay ASCII where they may reach a log (`->`, not an arrow glyph).
- Commit messages: a sentence in the repo's style, ending with `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`.

## Review Focus

```python
def test_round_trip_and_atomic_write(tmp_path: Path):
    clock = Clock()
    w = _writer(tmp_path, clock)
    w.start()
    hb = H.read(tmp_path / "hb.json")
    assert hb is not None and hb.run_id == "20260301T120000Z" and hb.pid == 4242 and hb.origin == "console"
    assert hb.started_at == "2026-03-01T12:00:00Z" and hb.updated_at == "2026-03-01T12:00:00Z"

    w.finish()

    clock = Clock()
    w = _writer(tmp_path, clock)
    w.start()
```

```python
"""The running sync's heartbeat: a small JSON file beside the database that says which run is in progress,
how far it is and who started it. `run_sync` keeps it current and removes it when it ends; `fdc console`
reads it, and a sync that wants its turn waits on it. Nothing private is in it."""
from __future__ import annotations

import json
import os
import sys
import time
from dataclasses import asdict, dataclass, field
from datetime import datetime, timedelta, timezone
from pathlib import Path
from typing import Callable

FILENAME = "sync.heartbeat.json"
MAX_AGE = timedelta(hours=3)          # the scheduled task's own execution limit is two
STAMP = "%Y-%m-%dT%H:%M:%SZ"


def stamp(dt: datetime) -> str:
    return dt.astimezone(timezone.utc).strftime(STAMP)


def parse_stamp(text: str) -> datetime | None:
    try:
        return datetime.strptime(text, STAMP).replace(tzinfo=timezone.utc)
    except (TypeError, ValueError):
        return None


@dataclass
class Heartbeat:
    run_id: str
    pid: int
    origin: str
    started_at: str
    updated_at: str
    step: str = ""
    detail: str = ""
    steps_done: list[str] = field(default_factory=list)

    def to_json(self) -> str:
        return json.dumps(asdict(self))

    @classmethod
    def from_json(cls, text: str) -> "Heartbeat | None":
        try:
            d = json.loads(text)
            return cls(run_id=str(d["run_id"]), pid=int(d["pid"]), origin=str(d["origin"]),
                       started_at=str(d["started_at"]), updated_at=str(d["updated_at"]),
                       step=str(d.get("step", "")), detail=str(d.get("detail", "")),
                       steps_done=[str(s) for s in d.get("steps_done", [])])
        except (ValueError, TypeError, KeyError, AttributeError):
            return None


def pid_alive(pid: int) -> bool:
    """Is a process with this id running? Never os.kill(pid, 0) on Windows: there it terminates the process."""
    if pid <= 0:
        return False
    if sys.platform == "win32":
        import ctypes
        from ctypes import wintypes

        PROCESS_QUERY_LIMITED_INFORMATION = 0x1000
        STILL_ACTIVE = 259
        kernel32 = ctypes.windll.kernel32
        handle = kernel32.OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION, False, pid)
        if not handle:
            return False
        try:
            code = wintypes.DWORD()
            if not kernel32.GetExitCodeProcess(handle, ctypes.byref(code)):
                return False
            return code.value == STILL_ACTIVE
        finally:
            kernel32.CloseHandle(handle)
    try:
        os.kill(pid, 0)
    except ProcessLookupError:
        return False
    except PermissionError:
        return True
    return True


def read(path: Path) -> Heartbeat | None:
    """The heartbeat on disk, or None when there is none or it is unreadable (half-written, malformed)."""
    try:
        return Heartbeat.from_json(Path(path).read_text(encoding="utf-8"))
    except OSError:
        return None


def is_alive(hb: Heartbeat, *, now: datetime, alive: Callable[[int], bool] = pid_alive,
             max_age: timedelta = MAX_AGE) -> bool:
    """Alive: its process runs and it was updated recently. Anything else is a run that died without cleaning up."""
    updated = parse_stamp(hb.updated_at)
    if updated is None or now - updated >= max_age:
        return False
    return alive(hb.pid)


class Writer:
    """Keeps one run's heartbeat current. Writes are atomic (temp file + replace) and throttled to one a second,
    except a step change, which always writes."""

    def __init__(self, path: Path, run_id: str, origin: str, *, pid: int | None = None,
                 now: Callable[[], datetime] = lambda: datetime.now(timezone.utc),
                 clock: Callable[[], float] = time.monotonic, min_interval: float = 1.0):
        self.path = Path(path)
        first = stamp(now())
        self.hb = Heartbeat(run_id=run_id, pid=pid if pid is not None else os.getpid(), origin=origin,
                            started_at=first, updated_at=first)
        self._now, self._clock, self._min = now, clock, min_interval
        self._last_write: float | None = None

    def start(self) -> None:
        self._write()

    def update(self, step: str, detail: str) -> None:
        changed = step != self.hb.step
        if changed and self.hb.step:
            self.hb.steps_done.append(self.hb.step)
        self.hb.step, self.hb.detail = step, detail
        if changed or self._last_write is None or self._clock() - self._last_write >= self._min:
            self._write()

    def finish(self) -> None:
        """Remove the file, unless another run has taken it over in the meantime."""
        current = read(self.path)
        if current is not None and current.pid != self.hb.pid:
            return
        try:
            self.path.unlink()
        except FileNotFoundError:
            pass

    def _write(self) -> None:
        self.hb.updated_at = stamp(self._now())
        self.path.parent.mkdir(parents=True, exist_ok=True)
        tmp = self.path.with_name(f"{self.path.name}.{self.hb.pid}.tmp")
        tmp.write_text(self.hb.to_json(), encoding="utf-8")
        os.replace(tmp, self.path)
        self._last_write = self._clock()


def wait_for_turn(path: Path, *, now: datetime, sleep: Callable[[float], None],
                  alive: Callable[[int], bool] = pid_alive, timeout: float = 1800.0, poll: float = 5.0) -> float:
    """Block while another sync's heartbeat is alive. Returns the seconds waited. Gives up after `timeout`
    seconds and lets the caller proceed: a wedged run must never stop syncing for good."""
    waited = 0.0
    while waited < timeout:
        hb = read(path)
        if hb is None or hb.pid == os.getpid() or not is_alive(hb, now=now + timedelta(seconds=waited), alive=alive):
            return waited
        sleep(poll)
        waited += poll
    return waited
```

```python
import shutil
import sqlite3

def _v7(tmp_path: Path) -> Path:
    older = tmp_path / "migrations"
    older.mkdir()
    for p in migrate.MIGRATIONS_DIR.glob("*.sql"):
        if int(p.name[:4]) < 8:
            shutil.copy(p, older)
    db = tmp_path / "w.db"
    conn = sqlite3.connect(db)
    migrate.apply_migrations(conn, migrations_dir=older)
    conn.execute("INSERT INTO sync_runs (run_id, step, started_at, finished_at, status, rows_written, message) "
                 "VALUES ('r1', 'prices', '2026-01-01T00:00:00Z', '2026-01-01T00:00:05Z', 'ok', 5, '')")
    conn.commit()
    assert conn.execute("SELECT MAX(version) FROM schema_version").fetchone()[0] == 7
    conn.close()
    return db
```

```python
    # ---- what the console shows -------------------------------------------
    def sync_status(self) -> list[sqlite3.Row]:
        """The last run of every step (the sync_status view)."""
        return self.query("SELECT * FROM sync_status ORDER BY step")

    def recent_run_rows(self, limit: int = 50) -> list[sqlite3.Row]:
        """Every step row of the newest `limit` runs, oldest row first (the caller groups by run_id)."""
        return self.query(
            """SELECT * FROM sync_runs
               WHERE run_id IN (SELECT run_id FROM (SELECT run_id, MAX(id) AS last_id FROM sync_runs
                                                    GROUP BY run_id ORDER BY last_id DESC LIMIT ?))
               ORDER BY id""", (limit,))

    def run_steps(self, run_id: str) -> list[sqlite3.Row]:
        return self.query("SELECT * FROM sync_runs WHERE run_id = ? ORDER BY id", (run_id,))

    def source_facts(self) -> dict:
        """Where the data came from, in one batch: ingested files per kind, price and SEC coverage, accounts per
        origin, and the active connections."""
        ingested = {r["kind"]: {"count": r["n"], "last": r["last"]} for r in self.query(
            "SELECT kind, COUNT(*) AS n, MAX(ingested_at) AS last FROM ingested_files GROUP BY kind")}
        p = self.query("SELECT COUNT(DISTINCT symbol) AS symbols, MAX(date) AS latest FROM prices")[0]
        s = self.query("SELECT COUNT(DISTINCT cik) AS companies, MAX(filed) AS latest_filed FROM sec_facts")[0]
        by_origin = {r["origin"]: r["n"] for r in self.query(
            "SELECT origin, COUNT(*) AS n FROM accounts GROUP BY origin")}
        return {
            "ingested": ingested,
            "prices": {"symbols": p["symbols"], "latest": p["latest"]},
            "sec": {"companies": s["companies"], "latest_filed": s["latest_filed"]},
            "accounts_by_origin": by_origin,
```

```python
def parse_iso(text: str | None) -> datetime | None:
    if not text:
        return None
    try:
        dt = datetime.fromisoformat(text.replace("Z", "+00:00"))
    except ValueError:
        return None
    return dt if dt.tzinfo else dt.replace(tzinfo=timezone.utc)


def seconds_between(a: str | None, b: str | None) -> float:
    da, db = parse_iso(a), parse_iso(b)
    return (db - da).total_seconds() if da and db else 0.0


def fmt_age(hours: float | None) -> str:
    if hours is None:
        return ""
    if hours < 1:
        return "<1 h ago"
    if hours < 48:
        return f"{hours:.0f} h ago"
    return f"{hours / 24:.1f} d ago"


def fmt_duration(seconds: float | None) -> str:
    if seconds is None:
        return ""
    s = int(round(seconds))
    if s < 60:
        return f"{s} s"
    if s < 3600:
        return f"{s // 60} min {s % 60:02d} s"
    return f"{s // 3600} h {(s % 3600) // 60:02d} min"


def local_time(iso: str | None, tz: tzinfo | None = None) -> str:
    dt = parse_iso(iso)
    if dt is None:
        return "never"
    return dt.astimezone(tz).strftime("%m-%d %H:%M")


# ---- assembling ---------------------------------------------------------------------------------------
def summarize_runs(rows: Iterable[Mapping]) -> list[RunSummary]:
    """Group step rows (oldest first) into runs, newest run first."""
    by_run: dict[str, list[Mapping]] = {}
    for r in rows:
        by_run.setdefault(r["run_id"], []).append(r)
    out = []
    for run_id, steps in by_run.items():
        status = {s["step"]: s["status"] for s in steps}
        started = min(s["started_at"] for s in steps)
        finished = max(s["finished_at"] for s in steps)
        out.append(RunSummary(
            run_id=run_id, origin=steps[0]["origin"], started_at=started, finished_at=finished,
            seconds=seconds_between(started, finished),
            statuses=tuple(status.get(step, "") for step in STEPS),
            rows=sum(int(s["rows_written"] or 0) for s in steps),
            errors=sum(1 for s in steps if s["status"] == "error"),
        ))
    out.reverse()
    return out


def _running(hb: H.Heartbeat | None, now: datetime, alive) -> Running:
    if hb is None:
        return Running("idle")
    started = parse_iso(hb.started_at)
    elapsed = (now - started).total_seconds() if started else None
    if H.is_alive(hb, now=now, alive=alive):
        return Running("running", hb, elapsed)
    return Running("stale", hb, elapsed)


def _step_state(step: str, row: Mapping | None, running: Running) -> StepState:
    marker = ""
    if running.state == "running" and running.heartbeat is not None:
        if running.heartbeat.step == step:
            marker = "running"
        elif step in running.heartbeat.steps_done:
            marker = "done"
    if row is None:
        return StepState(step, "never", None, False, 0, "", marker)
    age = row["age_hours"]
    stale = age is not None and age > STALE_HOURS.get(step, 48)
    return StepState(step, row["status"], age, stale, int(row["rows_written"] or 0), row["message"] or "", marker)


def _count(n: int, noun: str) -> str:
    return f"{n} {noun}{'' if n == 1 else 's'}"


def _ingested(ing: Mapping, prefix: str) -> str:
    items = [(v["last"], v["count"]) for k, v in ing.items() if k.startswith(prefix) and v["last"]]
    if not items:
        return "never"
    return f"{local_time(max(t for t, _ in items))}, {_count(sum(c for _, c in items), 'file')}"


def _sources(cfg: Config, facts: Mapping, status: Mapping[str, Mapping]) -> list[SourceRow]:
    rows: list[SourceRow] = []
    ing = facts["ingested"]
    if cfg.inbox.is_dir():
        waiting = [p for p in cfg.inbox.iterdir() if p.is_file()]
        state = f"{_count(len(waiting), 'file')} waiting" if waiting else "empty"
    else:
        state = "missing"
    rows.append(SourceRow("inbox files", str(cfg.inbox), _ingested(ing, "fidelity_"), state))
    if not cfg.snaptrade_enabled or cfg.snaptrade_dir is None:
        rows.append(SourceRow("broker export folder", str(cfg.snaptrade_dir or ""), "", "off"))
    else:
        rows.append(SourceRow("broker export folder", str(cfg.snaptrade_dir), _ingested(ing, "snaptrade_"),
                              "present" if cfg.snaptrade_dir.is_dir() else "missing"))
    for c in facts["connections"]:
        n = facts["accounts_by_origin"].get(c["name"], 0)
        state = f"error: {c['last_error']}" if c["last_error"] else ("ok" if c["last_ok_at"] else "never")
        rows.append(SourceRow(f"connection: {c['name']}", _count(n, "account"), local_time(c["last_ok_at"]), state))
    p, st = facts["prices"], status.get("prices")
    rows.append(SourceRow("prices", "Tiingo" if cfg.tiingo_token else "yfinance",
                          f"{_count(p['symbols'], 'symbol')}, latest {p['latest'] or 'none'}; "
                          f"step {local_time(st['finished_at']) if st else 'never'}",
                          st["status"] if st else "never"))
    s, st = facts["sec"], status.get("sec")
    state = "skipped: SEC_USER_AGENT not set" if not cfg.sec_user_agent else (st["status"] if st else "never")
    rows.append(SourceRow("SEC EDGAR", "companyfacts",

    if cfg.watchlist.is_file():
        rows.append(SourceRow("watchlist", str(cfg.watchlist), _count(len(read_watchlist(cfg.watchlist)), "symbol"), "present"))
    else:
        rows.append(SourceRow("watchlist", str(cfg.watchlist), "", "missing"))
    if cfg.investing_dir is None:
        rows.append(SourceRow("research tickers", "", "", "off"))
    else:
        rows.append(SourceRow("research tickers", str(cfg.investing_dir), "",
                              "present" if cfg.investing_dir.is_dir() else "missing"))
    st = status.get("export")
    if not cfg.export_cockpit:
        rows.append(SourceRow("-> cockpit feed files", str(cfg.export_dir or ""), "", "off"))
    else:
        folder_ok = cfg.export_dir is not None and cfg.export_dir.is_dir()
        state = (st["status"] if st else "never") if folder_ok else "missing folder"
        rows.append(SourceRow("-> cockpit feed files", str(cfg.export_dir),
                              f"{local_time(st['finished_at'])}: {st['message']}" if st else "never", state))

def load_overview(cfg: Config, *, now: datetime | None = None, store_open: Callable = Store.open,
                  heartbeat_read: Callable[[Path], H.Heartbeat | None] = H.read, alive=None,
                  schedule: Schedule | None = None, schedule_reason: str = "", runs_limit: int = 50) -> Overview:
    """One read of everything the screen shows. Opens the store without migrating and only reads."""
    now = now or datetime.now(timezone.utc)
    running = _running(heartbeat_read(cfg.heartbeat_path), now, alive)
    store = store_open(cfg.db_path, migrate=False)
    try:
        status = {r["step"]: r for r in store.sync_status()}
        facts = store.source_facts()
        runs = summarize_runs(store.recent_run_rows(runs_limit))
        universe = len(build_universe(store, cfg.watchlist, classify=cfg.classify, investing_dir=cfg.investing_dir))
        latest = store.latest_snapshot_date()
    finally:
        store.close()
    return Overview(
        root=cfg.root, now=now,
        steps=[_step_state(step, status.get(step), running) for step in STEPS],
        sources=_sources(cfg, facts, status), universe_size=universe, runs=runs, running=running,
        latest_snapshot=latest, schedule=schedule, schedule_reason=schedule_reason,
    )


def load_run_steps(cfg: Config, run_id: str, *, store_open: Callable = Store.open) -> list[RunStep]:
    store = store_open(cfg.db_path, migrate=False)
```

```python
_DURATION = re.compile(r"^P(?:(\d+)D)?(?:T(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?)?$")


def default_log_path(env: Mapping[str, str] = os.environ) -> Path | None:
    """Where scripts/run_sync.ps1 appends its log; None when the file does not exist."""
    base = env.get("LOCALAPPDATA")
    if not base:
        return None
    p = Path(base) / "financial-data-collector" / "sync.log"
    return p if p.is_file() else None


def _duration(text: str) -> str:
    m = _DURATION.match(text or "")
    if not m or not any(m.groups()):
        return ""
    days, hours, minutes, seconds = (int(g) if g else 0 for g in m.groups())
    parts = []
    if days:
        parts.append(f"{days} d")
    if hours:
        parts.append(f"{hours} h")
    if minutes:
        parts.append(f"{minutes} min")
    if seconds:
        parts.append(f"{seconds} s")
    return " ".join(parts)


def humanize_trigger(t: Mapping) -> str:
    kind = str(t.get("type", "")).replace("MSFT_Task", "").replace("Trigger", "")
    start = str(t.get("start") or "")
    clock = start[11:16] if len(start) >= 16 else ""
    if kind == "Logon":
        text = "at logon"
    elif kind == "Boot":
        text = "at startup"
    elif kind == "Daily":
        text = f"daily {clock}".strip()
    elif kind == "Weekly":
        text = f"weekly {clock}".strip()
    elif kind == "Time":
        text = f"once {start[:10]} {clock}".strip()
    else:
        text = kind or "trigger"
    delay = _duration(str(t.get("delay") or ""))
    if delay:
        text += f" +{delay}"
    interval = _duration(str(t.get("interval") or ""))
    if interval:
        text += f", every {interval}"
    if t.get("enabled") is False:
        text += " (disabled)"
    return text


def _date(text) -> str | None:
    if not text or not isinstance(text, str) or text[:4] < "2000":
        return None
    return text


def parse_tasks(text: str, log_path: Path | None) -> list[Schedule]:
    try:
        data = json.loads(text) if text.strip() else []
    except ValueError:
        return []
    if isinstance(data, dict):
        data = [data]
    if not isinstance(data, list):
        return []
    out = []
    for t in data:
        if not isinstance(t, dict):
            continue
        out.append(Schedule(
            name=str(t.get("name", "")), state=str(t.get("state", "")),
            last_run=_date(t.get("last_run")),
            last_result=int(t["last_result"]) if t.get("last_result") is not None else None,
            next_run=_date(t.get("next_run")),
            triggers=tuple(humanize_trigger(x) for x in t.get("triggers") or [] if isinstance(x, dict)),
            log_path=log_path,
        ))
    return out


def powershell_command(root: Path) -> str:
    needle = str(Path(root) / "scripts" / "run_sync.ps1").replace("'", "''")
    return (
        f"$needle = '{needle}'; "
        "$tasks = @(Get-ScheduledTask | Where-Object { (($_.Actions | ForEach-Object { $_.Arguments }) -join ' ') -like ('*' + $needle + '*') }); "
        "$out = @(foreach ($t in $tasks) { $i = $t | Get-ScheduledTaskInfo; [pscustomobject]@{ "
        "name = $t.TaskName; state = [string]$t.State; "
        "last_run = $(if ($i.LastRunTime) { $i.LastRunTime.ToString('s') } else { $null }); "
        "last_result = $i.LastTaskResult; "
        "next_run = $(if ($i.NextRunTime) { $i.NextRunTime.ToString('s') } else { $null }); "
        "triggers = @($t.Triggers | ForEach-Object { [pscustomobject]@{ type = $_.CimClass.CimClassName; "
        "start = [string]$_.StartBoundary; delay = [string]$_.Delay; interval = [string]$_.Repetition.Interval; "
        "enabled = [bool]$_.Enabled } }) } }); "
        "ConvertTo-Json -InputObject $out -Depth 5"
    )


def read_schedule(root: Path, *, run: Callable = subprocess.run, platform: str = sys.platform,
                  env: Mapping[str, str] = os.environ) -> ScheduleResult:
    """The sync task registered for this folder, found by the script its action runs."""
    if platform != "win32":
        return ScheduleResult(None, "no scheduler integration on this platform")
    try:
        done = run(["powershell.exe", "-NoProfile", "-NonInteractive", "-Command", powershell_command(root)],
                   capture_output=True, text=True, timeout=10)
    except subprocess.TimeoutExpired:
        return ScheduleResult(None, "scheduler: no answer in 10 s")
    except OSError as e:
        return ScheduleResult(None, f"scheduler: {e}")
    if done.returncode != 0:
        first = (done.stderr or done.stdout or "").strip().splitlines()
        return ScheduleResult(None, f"scheduler: {first[0] if first else 'exit ' + str(done.returncode)}")
    tasks = parse_tasks(done.stdout, default_log_path(env))
    if not tasks:
        return ScheduleResult(None, "no task runs scripts/run_sync.ps1 for this folder: register one with scripts/schedule_sync.ps1")
    return ScheduleResult(tasks[0], "")
```

```python
"""Starting a sync from the console: one `fdc` subprocess at a time, its output in data/console-last-run.log.
The launcher only starts and watches the process; what the run did is read back from the database and the
heartbeat like any other run. Quitting the console does not stop a launched sync."""
from __future__ import annotations

import subprocess
import sys
from dataclasses import dataclass
from datetime import datetime, timezone
from pathlib import Path
from typing import Callable

from ..heartbeat import Heartbeat

LOG_NAME = "console-last-run.log"


@dataclass(frozen=True)
class LaunchState:
    kind: str                           # idle | queued | running | finished | failed
    step: str | None = None             # None: the whole sync
    pid: int | None = None
    exit_code: int | None = None
    started_at: datetime | None = None
    finished_at: datetime | None = None
    error: str = ""

    @property
    def what(self) -> str:
        return "sync" if self.step is None else f"step {self.step}"


class Launcher:
    def __init__(self, root: Path, *, python: str = sys.executable, popen: Callable = subprocess.Popen,
                 now: Callable[[], datetime] = lambda: datetime.now(timezone.utc)):
        self.root = Path(root)
        self._python, self._popen, self._now = python, popen, now
        self._process = None
        self._step: str | None = None
        self._started_at: datetime | None = None
        self._finished_at: datetime | None = None
        self._failed = ""

    @property
    def log_path(self) -> Path:
        return self.root / "data" / LOG_NAME

    def command(self, step: str | None) -> list[str]:
        cmd = [self._python, "-m", "financial_data_collector", "--root", str(self.root), "sync", "--origin", "console"]
        if step:
            cmd += ["--only", step]
        return cmd

    @property
    def busy(self) -> bool:
        return self._process is not None and self._process.poll() is None

    def start(self, step: str | None = None) -> LaunchState:
        if self.busy:
            return LaunchState("failed", step, error="the console's own run is still going")
        self.clear()
        self._step, self._started_at = step, self._now()
        self.log_path.parent.mkdir(parents=True, exist_ok=True)
        try:
            log = open(self.log_path, "w", encoding="utf-8")
        except OSError as e:
            self._failed = f"cannot write {self.log_path}: {e}"
            return self.state(None)
        try:
            kwargs: dict = {"cwd": str(self.root), "stdout": log, "stderr": subprocess.STDOUT,
                            "stdin": subprocess.DEVNULL}
            if sys.platform == "win32":
                kwargs["creationflags"] = getattr(subprocess, "CREATE_NO_WINDOW", 0)
            else:
                kwargs["start_new_session"] = True
            self._process = self._popen(self.command(step), **kwargs)
        except OSError as e:
            self._failed = f"could not start: {e}"
            log.write(self._failed + "\n")
        finally:
            log.close()
        return self.state(None)

    def state(self, heartbeat: Heartbeat | None) -> LaunchState:
        if self._failed:
            return LaunchState("failed", self._step, started_at=self._started_at, error=self._failed)
        if self._process is None:
            return LaunchState("idle")
        code = self._process.poll()
        if code is None:
            running = heartbeat is not None and heartbeat.pid == self._process.pid
            return LaunchState("running" if running else "queued", self._step, self._process.pid,
                               started_at=self._started_at)
        if self._finished_at is None:
            self._finished_at = self._now()
        return LaunchState("finished", self._step, self._process.pid, code, self._started_at, self._finished_at)

    def clear(self) -> None:
        """Forget a finished or failed launch (a running one stays)."""
        if self.busy:
            return
        self._process, self._step, self._started_at, self._finished_at, self._failed = None, None, None, None, ""
```

```python
"""Rich renderables for an Overview: the one-shot `fdc console --once`, and the row builders the live screen
shares, so both look the same."""
from __future__ import annotations

from datetime import datetime, tzinfo

from rich.table import Table
from rich.text import Text

from .. import ui
from ..sync import STEPS
from .actions import LaunchState
from .model import (
    Overview, RunStep, RunSummary, SourceRow, StepState, fmt_age, fmt_duration, local_time, parse_iso,

from .schedule import Schedule

MESSAGE_WIDTH = 110
STATE_STYLES = {"ok": "green", "present": "green", "empty": "dim", "off": "dim", "never": "dim",
                "missing": "red", "missing folder": "red"}
LETTERS = {"ok": ("o", "green"), "skipped": ("s", "yellow"), "error": ("e", "red"), "dry-run": ("d", "cyan"),
           "": ("-", "dim")}
LEGEND = "steps in order: " + " ".join(STEPS) + "; o ok, s skipped, e error, - not run"


def state_text(state: str) -> Text:
    if state.startswith("error"):
        style = "red"
    elif state.startswith("skipped"):
        style = "yellow"
    elif state.endswith("waiting"):
        style = "dark_orange"
    else:
        style = STATE_STYLES.get(state, "")
    return Text(state, style=style)


def badges(statuses) -> Text:
    t = Text()
    for st in statuses:
        letter, style = LETTERS.get(st, ("?", ""))
        t.append(letter, style=style)
    return t


def step_row(s: StepState) -> list:
    status = Text(f"{s.status} STALE", style="dark_orange") if s.stale else ui.badge(s.status)
    now = {"running": Text("running", style="bold cyan"), "done": Text("done", style="green")}.get(s.now, Text(""))
    return [s.step, status, fmt_age(s.age_hours), s.rows, now, s.message[:MESSAGE_WIDTH]]


def source_row(s: SourceRow) -> list:
    return [s.name, s.where, s.last, state_text(s.state)]

    return [local_time(r.started_at, tz), r.origin, fmt_duration(r.seconds), badges(r.statuses), r.rows]

    return [s.step, ui.badge(s.status), fmt_duration(s.seconds), s.rows, s.message]


def status_text(ov: Overview, launch: LaunchState | None = None, now: datetime | None = None,
                tz: tzinfo | None = None) -> Text:
    now = now or ov.now
    r = ov.running
    t = Text()
    if r.state == "running" and r.heartbeat is not None:
        hb = r.heartbeat
        started = parse_iso(hb.started_at)
        elapsed = (now - started).total_seconds() if started else r.elapsed_seconds
        t.append("RUNNING", style="bold cyan")
        t.append(f"  {hb.step} · {hb.detail} · started {local_time(hb.started_at, tz)} by {hb.origin}"
                 f" · pid {hb.pid} · {fmt_duration(elapsed)}")
    elif r.state == "stale" and r.heartbeat is not None:
        hb = r.heartbeat
        t.append("DID NOT FINISH", style="bold red")
        t.append(f"  run {hb.run_id} started {local_time(hb.started_at, tz)} by {hb.origin} (pid {hb.pid} gone)")
    else:
        t.append("IDLE", style="bold green")
        if ov.runs:
            last = ov.runs[0]
            outcome = f"{last.errors} error{'s' if last.errors != 1 else ''}" if last.errors else "ok"
            t.append(f"  last run {local_time(last.started_at, tz)} ({last.origin}, {outcome})")
        else:
            t.append("  no runs yet")
    if launch is not None and launch.kind != "idle":
        if launch.kind == "queued":
            t.append(f"  ·  console {launch.what} queued", style="yellow")
        elif launch.kind == "running":
            t.append(f"  ·  console {launch.what}", style="cyan")
        elif launch.kind == "finished":
            when = launch.finished_at.astimezone(tz).strftime("%H:%M:%S") if launch.finished_at else ""
            t.append(f"  ·  FINISHED exit {launch.exit_code} · {when}",
                     style="green" if launch.exit_code == 0 else "red")
        elif launch.kind == "failed":
            t.append(f"  ·  LAUNCH FAILED · {launch.error}", style="red")
    return t


def _short(iso: str | None) -> str:
    return iso[5:16].replace("T", " ") if iso else ""


def schedule_text(schedule: Schedule | None, reason: str) -> Text:
    if schedule is None:
        return Text(f"schedule: {reason or 'unknown'}", style="dim")
    t = Text()
    t.append("task ", style="dim")
    t.append(schedule.name, style="bold")
    t.append(f"   state {schedule.state}   last {_short(schedule.last_run) if schedule.last_run else 'never'}")
    if schedule.last_run is not None and schedule.last_result is not None:
        t.append(f" (exit {schedule.last_result})", style="green" if schedule.last_result == 0 else "red")
    t.append(f"   next {_short(schedule.next_run) if schedule.next_run else 'none'}\n")
    t.append("triggers ", style="dim")
    t.append(" · ".join(schedule.triggers) or "none")
    if schedule.log_path is not None:
        t.append("\nlog ", style="dim")
        t.append(str(schedule.log_path))
    return t


def universe_text(ov: Overview) -> str:
    return f"following {ov.universe_size} symbol{'s' if ov.universe_size != 1 else ''}"


def steps_table(ov: Overview) -> Table:
```

```python
"""fdc console: the live screen. A renderer over console.model whose only side effect is starting `fdc sync`
through console.actions. Keys: 1/2/3 tabs, r run the sync, s run the step under the cursor, ctrl+p palette, q quit."""
from __future__ import annotations

from datetime import datetime, timedelta, timezone
from functools import partial
from pathlib import Path
from typing import Callable

from rich.text import Text
from textual.app import App, ComposeResult
from textual.binding import Binding
from textual.command import DiscoveryHit, Hit, Hits, Provider
from textual.containers import Horizontal
from textual.widgets import DataTable, Footer, Static, TabbedContent, TabPane
from textual.worker import Worker, WorkerState

from ..config import Config
from ..sync import STEPS
from . import report
from .actions import Launcher
from .model import Overview, RunStep, load_overview, load_run_steps
from .schedule import ScheduleResult, read_schedule

FINISHED_SHOWN_FOR = timedelta(seconds=10)
TAB_TABLES = {"overview": "#steps", "sources": "#sources_table", "runs": "#runs_table"}


class ConsoleCommands(Provider):
    """Palette entries: run the sync, run one step, refresh."""

    def _entries(self):
        app = self.app
        yield "Run sync", "run every step now", app.action_run_sync
        for step in STEPS:
            yield f"Run step: {step}", f"run only the {step} step", partial(app.run_step, step)
        yield "Refresh now", "reload the screen and the schedule", app.action_refresh

    async def discover(self) -> Hits:
        for name, help_, cmd in self._entries():
            yield DiscoveryHit(name, cmd, help=help_)

    async def search(self, query: str) -> Hits:
        matcher = self.matcher(query)
        for name, help_, cmd in self._entries():
            score = matcher.match(name)
            if score > 0:
                yield Hit(score, matcher.highlight(name), cmd, help=help_)


class ConsoleApp(App):
    TITLE = "fdc console"
    CSS = """
    #title { height: 1; padding: 0 1; color: $text-muted; }
    #status { height: 1; padding: 0 1; background: $surface; }
    #schedule { height: auto; padding: 1 1 0 1; }
    #universe { height: 1; padding: 0 1; }
    DataTable { height: 1fr; }
    #runs_table { width: 55%; }
    #run_detail { width: 45%; }
    """
    BINDINGS = [
        Binding("1", "tab('overview')", "Overview"),
        Binding("2", "tab('sources')", "Sources"),
        Binding("3", "tab('runs')", "Runs"),
        Binding("r", "run_sync", "Run sync"),
        Binding("s", "run_step_under_cursor", "Run step"),
        Binding("q", "quit", "Quit"),
    ]
    COMMANDS = App.COMMANDS | {ConsoleCommands}

    def __init__(self, cfg: Config, *, load: Callable[..., Overview] = load_overview,
                 load_steps: Callable[..., list[RunStep]] = load_run_steps, launcher=None,
                 schedule_reader: Callable[[Path], ScheduleResult] = read_schedule,
                 refresh_seconds: float = 2.0, schedule_seconds: float = 300.0,
                 now: Callable[[], datetime] = lambda: datetime.now(timezone.utc)):
        super().__init__()
        self.cfg = cfg
        self._load, self._load_steps, self._schedule_reader = load, load_steps, schedule_reader
        self.launcher = launcher if launcher is not None else Launcher(cfg.root)
        self._refresh_seconds, self._schedule_seconds, self._now = refresh_seconds, schedule_seconds, now
        self.overview: Overview | None = None
        self.schedule = ScheduleResult(None, "reading the scheduler...")
        self.refresh_error = ""
        self.selected_run: str | None = None
        self.status_text = Text("loading...")
        self.schedule_text = Text("")
        self.universe_line = ""

    # ---- layout --------------------------------------------------------------------------------------
    def compose(self) -> ComposeResult:
        yield Static(Text(f"fdc console  ·  {self.cfg.root}"), id="title")
        yield Static(self.status_text, id="status")
        with TabbedContent(initial="overview"):
            with TabPane("Overview", id="overview"):
                yield DataTable(id="steps", cursor_type="row", zebra_stripes=True)
                yield Static(self.schedule_text, id="schedule")
            with TabPane("Sources", id="sources"):
                yield Static("", id="universe")
                yield DataTable(id="sources_table", cursor_type="row", zebra_stripes=True)
            with TabPane("Runs", id="runs"):
                with Horizontal():
                    yield DataTable(id="runs_table", cursor_type="row", zebra_stripes=True)
                    yield DataTable(id="run_detail", cursor_type="none")
        yield Footer()

    def on_mount(self) -> None:
        self.query_one("#steps", DataTable).add_columns("step", "status", "age", "rows", "now", "message")
        self.query_one("#sources_table", DataTable).add_columns("source", "where / how", "last", "state")
        self.query_one("#runs_table", DataTable).add_columns("started", "origin", "took", "steps", "rows")
        self.query_one("#run_detail", DataTable).add_columns("step", "status", "took", "rows", "message")
        self._render_schedule()
        self.refresh_data()
        self.refresh_schedule()
        self.set_interval(self._refresh_seconds, self.refresh_data)
        self.set_interval(self._schedule_seconds, self.refresh_schedule)
        self.set_interval(1.0, self._render_status)

    # ---- background reads ----------------------------------------------------------------------------
    def refresh_data(self) -> None:
        cfg, sched, now = self.cfg, self.schedule, self._now()
        self.run_worker(lambda: self._load(cfg, now=now, schedule=sched.schedule, schedule_reason=sched.reason),
                        name="refresh", group="refresh", exclusive=True, thread=True, exit_on_error=False)

    def refresh_schedule(self) -> None:
        root = self.cfg.root
        self.run_worker(lambda: self._schedule_reader(root), name="schedule", group="schedule", exclusive=True,
                        thread=True, exit_on_error=False)

    def _load_detail(self, run_id: str) -> None:
        cfg = self.cfg
        self.run_worker(lambda: (run_id, self._load_steps(cfg, run_id)), name="steps", group="steps",
                        exclusive=True, thread=True, exit_on_error=False)

    def on_worker_state_changed(self, event: Worker.StateChanged) -> None:
        w = event.worker
        if w.state == WorkerState.SUCCESS:
            if w.name == "refresh":
                self.refresh_error = ""
                self._apply(w.result)
            elif w.name == "schedule":
                self.schedule = w.result
                self._render_schedule()
            elif w.name == "steps":
                self._fill_detail(*w.result)
        elif w.state == WorkerState.ERROR:
            if w.name == "refresh":
                self.refresh_error = f"refresh failed: {w.error}"
                self._render_status()
            elif w.name == "schedule":
                self.schedule = ScheduleResult(None, f"scheduler: {w.error}")
                self._render_schedule()

    # ---- rendering -----------------------------------------------------------------------------------
    def _apply(self, ov: Overview) -> None:
        self.overview = ov
        steps = self.query_one("#steps", DataTable)
        row = steps.cursor_row
        steps.clear()
        for s in ov.steps:
            steps.add_row(*report.step_row(s))
        steps.move_cursor(row=min(row, max(steps.row_count - 1, 0)))
        sources = self.query_one("#sources_table", DataTable)
        sources.clear()
        for s in ov.sources:
            sources.add_row(*report.source_row(s))
        self.universe_line = report.universe_text(ov)
        self.query_one("#universe", Static).update(self.universe_line)
        runs = self.query_one("#runs_table", DataTable)
        runs.clear()
        for r in ov.runs:
            runs.add_row(*report.run_row(r), key=r.run_id)
        if ov.runs:
            keys = [r.run_id for r in ov.runs]
            target = self.selected_run if self.selected_run in keys else keys[0]
            runs.move_cursor(row=keys.index(target))
            if target != self.selected_run or self.query_one("#run_detail", DataTable).row_count == 0:
                self.selected_run = target
                self._load_detail(target)
        self._render_status()

    def _fill_detail(self, run_id: str, steps: list[RunStep]) -> None:
        if run_id != self.selected_run:
            return
        detail = self.query_one("#run_detail", DataTable)
        detail.clear()
        for s in steps:
            detail.add_row(*report.run_step_row(s))

    def _render_status(self) -> None:
        if self.overview is None:
            text = Text("loading...")
        else:
            now = self._now()
            state = self.launcher.state(self.overview.running.heartbeat)
            if state.kind == "finished" and state.finished_at and now - state.finished_at > FINISHED_SHOWN_FOR:
                self.launcher.clear()
                state = self.launcher.state(self.overview.running.heartbeat)
            text = report.status_text(self.overview, state, now=now)
        if self.refresh_error:
            text.append(f"  ·  {self.refresh_error}", style="red")
        self.status_text = text
        self.query_one("#status", Static).update(text)

    def _render_schedule(self) -> None:
        self.schedule_text = report.schedule_text(self.schedule.schedule, self.schedule.reason)
        self.query_one("#schedule", Static).update(self.schedule_text)

    # ---- actions -------------------------------------------------------------------------------------
    def action_tab(self, tab: str) -> None:
        self.query_one(TabbedContent).active = tab
        self.query_one(TAB_TABLES[tab], DataTable).focus()

    def action_run_sync(self) -> None:
        self.run_step(None)

    def action_run_step_under_cursor(self) -> None:
        steps = self.query_one("#steps", DataTable)
        if steps.row_count:
            self.run_step(STEPS[steps.cursor_row])

    def action_refresh(self) -> None:
        self.refresh_data()
        self.refresh_schedule()

    def run_step(self, step: str | None) -> None:
        if self.overview is not None and self.overview.running.state == "running":
            self.notify("a sync is running; wait for it to finish", severity="warning")
            return
        if self.launcher.busy:
            self.notify("the console's own run is still going", severity="warning")
            return
        state = self.launcher.start(step)
        if state.kind == "failed":
            self.notify(state.error, severity="error")
        else:
            self.notify(f"started {state.what}")
        self._render_status()

    def on_data_table_row_highlighted(self, event: DataTable.RowHighlighted) -> None:
        if event.data_table.id != "runs_table" or event.row_key is None:
            return
        run_id = event.row_key.value
        if run_id and run_id != self.selected_run:
            self.selected_run = run_id
            self._load_detail(run_id)
```
